210 template <class T>
211 [[nodiscard]] T getProperty(cstring propertyName) const {
212 auto iterator = stateProperties.find(propertyName);
213 if (iterator != stateProperties.end()) {
214 auto val = iterator->second;
215 try {
216 T resolvedVal = std::get<T>(val);
217 return resolvedVal;
218 } catch (std::bad_variant_access const &ex) {
219 BUG("Expected property value type does not correspond to value type stored in the "
220 "property map.");
221 }
222 }
223 BUG("Property %s not found in configuration map.", propertyName);
224 }

284 //
285 // Needs to be implemented here because of limitations with templates.
286 template <class N>
287 void replaceTopBody(const std::vector<const N *> *nodes) {
288 BUG_CHECK(!nodes->empty(), "Replaced top of execution stack with empty list");
289 body.pop();
290 for (auto it = nodes->rbegin(); it != nodes->rend(); ++it) {
291 BUG_CHECK(*it, "Evaluation produced a null node.");
292 body.push(*it);
293 }
294 }
