25/* iterate over ranges of contiguous bits in a bitvector */
26class bitranges {
27 bitvec tmp;
28 const bitvec &bits;
29 struct iter {
30 bool valid = true;
31 bitvec::const_bitref ptr;
32 std::pair<int, int> range;
33
34 iter &operator++() {
35 if (ptr) {
36 range.first = range.second = ptr.index();
37 while (++ptr && range.second + 1 == ptr.index()) ++range.second;
38 } else {
39 valid = false;
40 }
41 return *this;
42 }
43 std::pair<int, int> operator*() { return range; }
44 bool operator==(iter &a) const { return valid == a.valid && ptr == a.ptr; }
45 bool operator!=(iter &a) const { return !(*this == a); }
46 explicit iter(bitvec::const_bitref p) : ptr(p) { ++*this; }
47 };
48
49 public:
50 explicit bitranges(const bitvec &b) : bits(b) {}
51 explicit bitranges(bitvec &&b) : tmp(b), bits(tmp) {}
52 explicit bitranges(uintptr_t b) : tmp(b), bits(tmp) {}
53 iter begin() const { return iter(bits.begin()); }
54 iter end() const { return iter(bits.end()); }
55};

69constexpr inline int divideFloor(int dividend, int divisor) {
70#if defined(__GNUC__) || defined(__clang__)
71 // Code to enable compiler folding when the divisor is a power-of-two compile-time constant
72 // In this case, compiler should fold to a right-shift
73 // FIXME: Replace absl with std after moving to C++20
74 unsigned u_divisor = static_cast<unsigned>(divisor);
75 if (__builtin_constant_p(u_divisor) && absl::has_single_bit(u_divisor))
76 return dividend >> (absl::bit_width(u_divisor) - 1);
77#endif // defined(__GNUC__) || defined(__clang__)
78
79 const int quotient = dividend / divisor;
80 const int remainder = dividend % divisor;
81 if ((remainder != 0) && ((remainder < 0) != (divisor < 0))) return quotient - 1;
82 return quotient;
83}
84
92constexpr int modulo(int dividend, int divisor) {
93 return (dividend % divisor) * ((dividend < 0) != (divisor < 0) ? -1 : 1);
94}
95
109constexpr inline int moduloFloor(const int dividend, const int divisor) {
110#if defined(__GNUC__) || defined(__clang__)
111 // Code to enable compiler folding when the divisor is a power-of-two compile-time constant
112 // In this case, compiler should fold to a bitwise-and
113 // FIXME: Replace absl with std after moving to C++20
114 if (__builtin_constant_p(divisor) && absl::has_single_bit(static_cast<unsigned>(divisor)))
115 return dividend & (divisor - 1);
116#endif // defined(__GNUC__) || defined(__clang__)
117
118 const int remainder = modulo(dividend, divisor);
119 if (remainder == 0 || dividend >= 0) return remainder;
120 return divisor - remainder;
121}

707template <RangeUnit Unit, Endian Order>
708std::pair<HalfOpenRange<Unit, Order>, HalfOpenRange<Unit, Order>> operator-(
709 HalfOpenRange<Unit, Order> left, HalfOpenRange<Unit, Order> right) {
710 HalfOpenRange<Unit, Order> empty = {0, 0};
711 HalfOpenRange<Unit, Order> intersect = left.intersectWith(right);
712 if (intersect.empty())
713 return left.lo < right.lo ? std::make_pair(left, empty) : std::make_pair(empty, left);
714
715 HalfOpenRange<Unit, Order> lower =
716 left.lo == intersect.lo ? empty : BitRange::FromTo(left.lo, intersect.lo - 1);
717 HalfOpenRange<Unit, Order> upper =
718 left.hi == intersect.hi ? empty : BitRange::FromTo(intersect.hi, left.hi - 1);
719
720 return {lower, upper};
721}
